cmake_minimum_required(VERSION 3.16)
project(yolo26_lite LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)
add_library(yolo26_core software/src/postprocess.cpp software/src/frames.cpp software/src/preprocess.cpp
                       software/src/control.cpp software/src/compose.cpp software/src/metrics.cpp
                       software/src/display_buffers.cpp software/src/display_control.cpp
                       software/src/inference_completion.cpp software/src/video_engine.cpp software/src/npu_trace.cpp)
target_include_directories(yolo26_core PUBLIC software/include)
target_link_libraries(yolo26_core PUBLIC Threads::Threads)
option(YOLO26_FORCE_SCALAR "Disable AArch64 NEON CPU kernels for comparison" OFF)
if(YOLO26_FORCE_SCALAR)
    target_compile_definitions(yolo26_core PRIVATE YOLO26_FORCE_SCALAR=1)
endif()
add_executable(yolo26_cpu_benchmark software/src/cpu_benchmark.cpp)
target_link_libraries(yolo26_cpu_benchmark PRIVATE yolo26_core)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(yolo26_core PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
endif()

if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    add_library(yolo26_capture software/src/capture.cpp software/src/capture_color.cpp)
    target_link_libraries(yolo26_capture PUBLIC yolo26_core)
    target_compile_options(yolo26_capture PRIVATE -Wall -Wextra -Wpedantic)
    add_executable(yolo26_camera_probe software/src/camera_probe.cpp)
    target_link_libraries(yolo26_camera_probe PRIVATE yolo26_capture)
endif()

option(YOLO26_BUILD_SDK "Build the explicitly invoked Icraft runtime/probe (Linux only)" OFF)
if(YOLO26_BUILD_SDK)
    if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
        message(FATAL_ERROR "The qualified SDK runtime targets Linux ARM64")
    endif()
    set(YOLO26_RELEASE_GUARD_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../software/pose180_pipeline/src/inference"
        CACHE PATH "Directory containing the existing shared ReleaseGuard header")
    if(NOT EXISTS "${YOLO26_RELEASE_GUARD_DIR}/release_guard.hpp")
        message(FATAL_ERROR "The canonical existing ReleaseGuard source is required")
    endif()
    find_package(icraft-hostbackend 3.39.0.0 EXACT REQUIRED)
    find_package(icraft-zg330backend 3.39.0.0 EXACT REQUIRED)
    add_library(yolo26_icraft software/src/icraft_runtime.cpp software/src/icraft_video_io.cpp
                             software/src/icraft_diagnostics.cpp software/src/icraft_tensor_io.cpp)
    set_target_properties(yolo26_icraft PROPERTIES CXX_EXTENSIONS ON)
    target_include_directories(yolo26_icraft PRIVATE "${YOLO26_RELEASE_GUARD_DIR}")
    target_compile_options(yolo26_icraft PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
    target_link_libraries(yolo26_icraft PUBLIC yolo26_core Icraft::HostBackend Icraft::ZG330Backend)
    target_link_options(yolo26_icraft INTERFACE -Wl,--no-as-needed)
    add_executable(yolo26_infer_probe software/src/infer_probe.cpp)
    set_target_properties(yolo26_infer_probe PROPERTIES CXX_EXTENSIONS ON)
    target_link_libraries(yolo26_infer_probe PRIVATE yolo26_icraft)
    add_library(yolo26_live software/src/live_pipeline.cpp)
    target_link_libraries(yolo26_live PUBLIC yolo26_icraft yolo26_capture)
    target_compile_options(yolo26_live PRIVATE -Wall -Wextra -Wpedantic -ffp-contract=off)
    target_link_libraries(yolo26_infer_probe PRIVATE yolo26_live)
endif()
